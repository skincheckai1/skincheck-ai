            if 'gemini-1.5-flash' in m:

@app.post("/analyze")
async def analyze_face_and_suggest(file: UploadFile = File(...)):
    try:
        # 1. Leggiamo l'immagine correttamente per Gemini
        image_data = await file.read()
        image = Image.open(io.BytesIO(image_data))
        
        # 2. Il Prompt "Beauty Advisor" (Quello che ti piaceva e funzionava)
        prompt = """
        Agisci come un'esperta Personal Beauty Advisor in una profumeria di lusso.
        Analizza attentamente la foto del viso di questa cliente.
        
        Sulla base di ciò che vedi (solo a scopo estetico), fornisci:
        1. **La Tua Impressione Estetica**: Descrivi la pelle con gentilezza (es. "Luminosa", "Zona T lucida").
        2. **Beauty Routine Su Misura**: Crea una routine di 3 passaggi (Detergente, Siero, Crema).
        3. **Suggerimenti di Shopping (Marca, Prodotto, Prezzo)**: 
           Suggerisci prodotti reali (es. Clarins, Kiehl's, La Roche-Posay) con il **Prezzo Indicativo** (es. ~€30).

        Usa emoji, un tono elegante e formatta bene in grassetto. Rispondi in italiano.
        """
        
        # 3. Generazione del contenuto
        response = model.generate_content([prompt, image])
        
        if not response.text:
            return JSONResponse(content={"result": "L'IA non ha prodotto testo. Riprova."}, status_code=200)

        return JSONResponse(content={"result": response.text})

    except Exception as e:
        print(f"❌ Errore durante l'analisi: {str(e)}")
        return JSONResponse(content={"result": f"Errore: {str(e)}"}, status_code=500)
